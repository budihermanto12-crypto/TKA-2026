function doPost(e) {
  try {
    var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
    var data = JSON.parse(e.postData.contents);
    
    // Buat Header otomatis jika sheet masih kosong
    if (sheet.getLastRow() === 0) {
      sheet.appendRow(["Tanggal & Waktu", "Nama Siswa", "No Absen / Kelas", "Skor Akhir", "Status Kelulusan", "Jumlah Benar", "Jumlah Salah"]);
      sheet.getRange(1, 1, 1, 7).setFontWeight("bold").setBackground("#e2e8f0");
    }
    
    sheet.appendRow([
      data.timestamp || new Date().toLocaleString('id-ID'),
      data.name || "-",
      data.studentNo || "-",
      data.score !== undefined ? data.score : 0,
      data.status || "-",
      data.correct !== undefined ? data.correct : 0,
      data.wrong !== undefined ? data.wrong : 0
    ]);
    
    return ContentService.createTextOutput(JSON.stringify({result: "success"}))
      .setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({result: "error", error: err.toString()}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {

}
